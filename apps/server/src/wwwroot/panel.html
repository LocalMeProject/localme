import 'dart:async';
import 'dart:convert';
import 'dart:math';
import 'package:flutter/widgets.dart';
import 'package:web_socket_channel/io.dart';
import 'package:web_socket_channel/web_socket_channel.dart';

enum RealtimeStatus { disconnected, connecting, connected, reconnecting }

class LocalMeRealtimeManager with WidgetsBindingObserver {
  final String projectId;
  final String baseUrl;
  final Set<String> _subscribedTables = {};
  final _eventController = StreamController<Map<String, dynamic>>.broadcast();
  final _statusController = StreamController<RealtimeStatus>.broadcast();

  WebSocketChannel? _channel;
  Timer? _heartbeatTimer;
  int _retryCount = 0;
  bool _isDisposed = false;

  LocalMeRealtimeManager({required this.projectId, this.baseUrl = 'ws://api.localme.ir'}) {
    WidgetsBinding.instance.addObserver(this);
    connect();
  }

  Stream<Map<String, dynamic>> get events => _eventController.stream;
  Stream<RealtimeStatus> get status => _statusController.stream;

  void connect() {
    if (_isDisposed) return;
    _statusController.add(_retryCount == 0 ? RealtimeStatus.connecting : RealtimeStatus.reconnecting);

    final uri = Uri.parse('$baseUrl/_baas/v1/$projectId/realtime');
    try {
      _channel = IOWebSocketChannel.connect(
        uri,
        pingInterval: const Duration(seconds: 25), // حل مشکل Carrier NAT Timeout
      );

      _channel!.stream.listen(
        (message) {
          _retryCount = 0;
          _statusController.add(RealtimeStatus.connected);
          final data = jsonDecode(message as String) as Map<String, dynamic>;
          if (data['action'] == 'CONNECTED') {
            _resubscribeAll();
            return;
          }
          _eventController.add(data);
        },
        onError: (err) => _scheduleReconnect(),
        onDone: () => _scheduleReconnect(),
      );

      _startHeartbeat();
    } catch (_) {
      _scheduleReconnect();
    }
  }

  void subscribe(String table) {
    _subscribedTables.add(table);
    if (_channel != null) {
      _channel!.sink.add(jsonEncode({'action': 'subscribe', 'table': table}));
    }
  }

  void _resubscribeAll() {
    for (final tbl in _subscribedTables) {
      _channel?.sink.add(jsonEncode({'action': 'subscribe', 'table': tbl}));
    }
  }

  void _startHeartbeat() {
    _heartbeatTimer?.cancel();
    _heartbeatTimer = Timer.periodic(const Duration(seconds: 25), (_) {
      _channel?.sink.add(jsonEncode({'action': 'ping'}));
    });
  }

  void _scheduleReconnect() {
    _heartbeatTimer?.cancel();
    _channel = null;
    if (_isDisposed) return;

    _statusController.add(RealtimeStatus.reconnecting);
    // Exponential backoff: 1s, 2s, 4s, 8s, up to 30s max
    final delaySeconds = min(30, pow(2, _retryCount).toInt());
    _retryCount++;
    Timer(Duration(seconds: delaySeconds), () => connect());
  }

  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    // قطع کانکشن در پس زمینه برای کاهش مصرف باتری، اتصال مجدد در فورگراند
    if (state == AppLifecycleState.paused) {
      _channel?.sink.close();
      _heartbeatTimer?.cancel();
    } else if (state == AppLifecycleState.resumed) {
      _retryCount = 0;
      connect();
    }
  }

  void dispose() {
    _isDisposed = true;
    WidgetsBinding.instance.removeObserver(this);
    _heartbeatTimer?.cancel();
    _channel?.sink.close();
    _eventController.close();
    _statusController.close();
  }
}

package ir.localme.realtime

import kotlinx.coroutines.*
import kotlinx.coroutines.flow.*
import okhttp3.*
import org.json.JSONObject
import java.util.concurrent.TimeUnit

class LocalMeRealtimeClient(
    private val projectId: String,
    private val client: OkHttpClient = OkHttpClient.Builder()
        .pingInterval(25, TimeUnit.SECONDS)
        .retryOnConnectionFailure(true)
        .build()
) {
    private val _events = MutableSharedFlow<JSONObject>(extraBufferCapacity = 64)
    val events: SharedFlow<JSONObject> = _events.asSharedFlow()

    private var webSocket: WebSocket? = null
    private val subscribedTables = mutableSetOf<String>()

    fun connect() {
        val request = Request.Builder()
            .url("ws://api.localme.ir/_baas/v1/$projectId/realtime")
            .build()

        webSocket = client.newWebSocket(request, object : WebSocketListener() {
            override fun onOpen(ws: WebSocket, response: Response) {
                // عضویت مجدد در تمامی جداول
                subscribedTables.forEach { table ->
                    ws.send(JSONObject().apply {
                        put("action", "subscribe")
                        put("table", table)
                    }.toString())
                }
            }

            override fun onMessage(ws: WebSocket, text: String) {
                val json = JSONObject(text)
                _events.tryEmit(json)
            }

            override fun onFailure(ws: WebSocket, t: Throwable, response: Response?) {
                // تلاش مجدد بعد از ۳ ثانیه
                CoroutineScope(Dispatchers.IO).launch {
                    delay(3000)
                    connect()
                }
            }
        })
    }

    fun subscribe(table: String) {
        subscribedTables.add(table)
        webSocket?.send(JSONObject().apply {
            put("action", "subscribe")
            put("table", table)
        }.toString())
    }
}

import { useEffect, useRef, useState, useCallback } from 'react';
import { AppState, AppStateStatus } from 'react-native';

export function useRealtimeTable<T = any>(projectId: string, table: string) {
  const [data, setData] = useState<T[]>([]);
  const [isConnected, setIsConnected] = useState(false);
  const wsRef = useRef<WebSocket | null>(null);
  const retryRef = useRef(0);

  const connect = useCallback(() => {
    const ws = new WebSocket(`ws://api.localme.ir/_baas/v1/${projectId}/realtime`);

    ws.onopen = () => {
      setIsConnected(true);
      retryRef.current = 0;
      ws.send(JSON.stringify({ action: 'subscribe', table }));
    };

    ws.onmessage = (event) => {
      const msg = JSON.parse(event.data);
      if (msg.table !== table) return;

      if (msg.action === 'INSERT') {
        setData((prev) => [msg.data, ...prev]);
      } else if (msg.action === 'UPDATE') {
        setData((prev) => prev.map((item: any) => item.id === msg.data.id ? msg.data : item));
      } else if (msg.action === 'DELETE') {
        setData((prev) => prev.filter((item: any) => item.id !== msg.data.id));
      }
    };

    ws.onclose = () => {
      setIsConnected(false);
      const delay = Math.min(30000, 1000 * Math.pow(2, retryRef.current));
      retryRef.current += 1;
      setTimeout(connect, delay);
    };

    wsRef.current = ws;
  }, [projectId, table]);

  useEffect(() => {
    connect();
    // مدیریت قطع در بک گراند موبایل
    const sub = AppState.addEventListener('change', (state: AppStateStatus) => {
      if (state === 'active') connect();
      else wsRef.current?.close();
    });

    return () => {
      sub.remove();
      wsRef.current?.close();
    };
  }, [connect]);

  return { data, isConnected };
}
